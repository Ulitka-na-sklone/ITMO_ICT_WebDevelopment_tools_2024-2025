def worker(_):
    return fetch_sync()


def run_multiprocessing():
    with ProcessPoolExecutor(max_workers=5) as executor:
        results = executor.map(worker, range(5))
        all_users = []
        for r in results:
            all_users.extend(r)
        save_users(all_users)

Многопоточность

def save_users(usernames):
    with Session(engine) as session:
        for name in usernames:
            user = User(username=name, email=f"{name}@mail.com", hashed_password="hashed")
            session.add(user)
        session.commit()


def run_threading():
    with ThreadPoolExecutor(max_workers=5) as executor:
        futures = [executor.submit(fetch_sync) for _ in range(5)]
        results = []
        for f in futures:
            results.extend(f.result())
        save_users(results)

Общее время выполнения: 1.17 секунд

Асинхронность

async def run_async():
    tasks = [fetch_async() for _ in range(5)]
    results = await asyncio.gather(*tasks)
    all_users = []
    for r in results:
        all_users.extend(r)
    async with AsyncSessionLocal() as session:
        for name in all_users:
            user = User(username=name, email=f"{name}@mail.com", hashed_password="hashed")
            session.add(user)
        await session.commit()

Общее время выполнения: 0.56 секунд

Вывод

Наиболее выгодным вариантом является асинхронное выполнение, так как между получением ответа от сервера есть задержка, которая используется, чтобы продолжить работу. Далее идет многопоточность, которая страдает из-за задержек в получении данных, но работает достаточно быстро, так как потоки выполняются одновременно. Мультипроцессинг является самым медленным, так как он и страдает из-за задержек в получении данных, и имеет большие расходы на создание процесса.
